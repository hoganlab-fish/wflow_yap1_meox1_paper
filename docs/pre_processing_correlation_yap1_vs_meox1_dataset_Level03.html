run_cluster_correlation <- function(seurat_yap = list_objects$D10025_yap1_dataset_Level_03,
                                    seurat_meox = list_objects$D10051_meox1_dataset_Level_03_stringent,
                                    resolution_yap = "RNA_snn_res.0.1",
                                    resolution_meox = "RNA_snn_res.0.1"){
  #get GEX
  Expression_assay_yap <- GetAssayData(seurat_yap, layer = "scale.data", assay = "RNA")
  Expression_assay_meox <- GetAssayData(seurat_meox, layer = "scale.data", assay = "RNA")
  #intersect of genes
  genes <- intersect(rownames(Expression_assay_yap), rownames(Expression_assay_meox))
  Expression_assay_yap <- Expression_assay_yap[genes,]
  Expression_assay_meox <- Expression_assay_meox[genes,]
  
  #get cluster info
  clusters_yap <- seurat_yap@meta.data %>% 
      pull(resolution_yap) %>% unique()
  clusters_meox <- seurat_meox@meta.data %>% 
      pull(resolution_meox) %>% unique()
  
  #initiate matrices
  corr_mat <- pval_mat <- matrix(data = NA, nrow = length(clusters_yap), ncol = length(clusters_meox))
  colnames(corr_mat) <- colnames(pval_mat) <- clusters_meox
  rownames(corr_mat) <- rownames(pval_mat) <- clusters_yap
    
    #Populate correlation matrix
  for (cluster1 in clusters_yap){
    for (cluster2 in clusters_meox){
      #get barcodes
      barcodes1 <- seurat_yap@meta.data %>% 
        rownames_to_column("Barcode") %>% 
        dplyr::rename(chosen_res=resolution_yap) %>% 
        filter(., chosen_res == cluster1) %>% pull(Barcode)
      barcodes2 <- seurat_meox@meta.data %>% 
        rownames_to_column("Barcode") %>% 
        dplyr::rename(chosen_res=resolution_meox) %>% 
        filter(., chosen_res == cluster2) %>% pull(Barcode)
        
      #filter data sets
      expr_yap <- Expression_assay_yap[, colnames(Expression_assay_yap) %in% barcodes1] %>% as.data.frame() %>% 
        apply(., 1, mean)
      expr_meox <- Expression_assay_meox[, colnames(Expression_assay_meox) %in% barcodes2] %>% as.data.frame() %>% 
        apply(., 1, mean)

      #calculate correlation
        cor_res <- cor.test(expr_yap, expr_meox, method = 'spearman')
        corr_mat[cluster1, cluster2] <- cor_res$estimate
        pval_mat[cluster1, cluster2] <- cor_res$p.value
        
      } 
    
  }

  return(list(corr=corr_mat, pval=pval_mat))

}

run_correlation_plots <- function(list_in){
    #get correlation ready
    plot_df_corr <- list_in$corr %>% as.data.frame() %>%
      rownames_to_column("cluster1") %>% 
    pivot_longer(!cluster1, names_to = "cluster2", values_to = "Correlation") %>% 
      mutate(., cluster1= factor(cluster1, levels = as.character(seq(0,100,1))), #overshoot
                cluster2= factor(cluster2, levels = as.character(seq(0,100,1))))
    #get pvals ready
    plot_df_pval <- list_in$pval %>% as.data.frame() %>%
      rownames_to_column("cluster1") %>% 
    pivot_longer(!cluster1, names_to = "cluster2", values_to = "p-value") %>% 
      mutate(., cluster1= factor(cluster1, levels = as.character(seq(0,100,1))), #overshoot
                cluster2= factor(cluster2, levels = as.character(seq(0,100,1))))
    plot_df <- full_join(plot_df_corr,plot_df_pval) %>% 
      mutate(., signs = ifelse(`p-value` < 0.05, round(Correlation, 2), "x"))
    
    #make plot
    heatmap.correlation <- ggplot(plot_df, 
                                  aes(x = cluster1, y = cluster2, fill = Correlation)) +
      geom_tile() + theme_bw() +
      scale_fill_gradient2(high = '#762A83', low = "#1B7837", mid = 'white', limits =c(-1,1)) +
      geom_text(aes(label = signs)) +
      labs(fill = 'Correlation Coefficient') +
      plot_sizing_keeplegend +
      xlab("yap1 clusters") + ylab("meox1 clusters")
    
    return(heatmap.correlation)
  
}

                              score, 
                              celltype_order){
  
  #new labels for x axis
  new_labels <- df_in %>% 
  select(., score, L3_celltype, Genotype) %>% 
  mutate(., L3_celltype = factor(L3_celltype, levels = celltype_order)) %>% 
  group_by(Genotype, L3_celltype) %>% 
  summarise(., count = n()) %>% 
  group_by(L3_celltype) %>% 
  summarise(., label = paste0(L3_celltype,"\nwt=", count[1] , ", mut=",count[2])) %>% 
  pull(label) %>% unique()
  

  #rename
  colnames(df_in)[colnames(df_in) == score] <- "score"
  
  #y limits for test
  y_max <- df_in$score %>% max()
  y_pairwise <- y_max + (y_max*0.2)
  y_global <- y_max + (y_max*0.4)
  
  #get stats
  stat_list <- list()
  stat_list[["global"]] <- compare_means(score ~ Genotype,
                               data = df_in, 
                               method = "kruskal.test")

run_boxplot_stats_RAW_pval <- function(df_in= level_03@meta.data,
                              score, 
                              celltype_order){
  
  #new labels for x axis
  new_labels <- df_in %>% 
  select(., score, L3_celltype, Genotype) %>% 
  mutate(., L3_celltype = factor(L3_celltype, levels = celltype_order)) %>% 
  group_by(Genotype, L3_celltype) %>% 
  summarise(., count = n()) %>% 
  group_by(L3_celltype) %>% 
  summarise(., label = paste0(L3_celltype,"\nwt=", count[1] , ", mut=",count[2])) %>% 
  pull(label) %>% unique()
  

  #rename
  colnames(df_in)[colnames(df_in) == score] <- "score"
  
  #y limits for test
  y_max <- df_in$score %>% max()
  y_pairwise <- y_max + (y_max*0.2)
  y_global <- y_max + (y_max*0.4)
  
  #get stats
  stat_list <- list()
  stat_list[["global"]] <- compare_means(score ~ Genotype,
                               data = df_in, 
                               method = "kruskal.test")
  stat_list[["pairwise"]] <- compare_means(score ~ Genotype,
                                 data =df_in, 
                                 group.by = "L3_celltype", 
                                 method = "wilcox.test")
  
  plot <- df_in %>% 
    select(., score, L3_celltype, Genotype) %>% 
  mutate(., L3_celltype = factor(L3_celltype, levels = celltype_order),
            Genotype = factor(Genotype, levels = c("wildtype", "yap1_mutant"))) %>% 
  ggplot(.) +
  geom_boxplot(aes(x = L3_celltype, y = score, fill = Genotype), 
               position = position_dodge2(),outlier.shape = NA) +
  scale_fill_manual(values = genotype_colours) +
  theme_bw(base_size = 14)+ 
  theme(axis.text.x = element_text(angle = 90, vjust = 0.5, hjust=1, size=15)) +
  xlab("") + ylab(paste0(score, " score")) +
  geom_jitter(aes(x = L3_celltype, y = score, fill = Genotype),
              size = 0.1, alpha = 0.5, 
              position  = position_jitterdodge(jitter.width = 0.3, dodge.width = 0.75)) +
  geom_text(data = stat_list$pairwise, aes(x = L3_celltype, y = y_pairwise, label = p.signif)) +
  geom_text(data = stat_list$global, aes(x = 2.5, y = y_global, label = paste0(method, ", p = ", p.adj))) +
  scale_x_discrete(labels= new_labels) +
  ggtitle(paste0(score, ", RAW PVALUE"))
  
  
  return(plot)

}
